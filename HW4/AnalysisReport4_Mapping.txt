# Step 4-5: Read Groups & Marking Duplicates, re-running Qualimap
#   Let's try to run picard without adding read groups and hope that it works. It did not work so we need to add read groups. uggghh... Added these to a script.. sooo many scripts... :'(
       picard AddOrReplaceReadGroups I=22018FL-04-01-12_S12_L002_sorted.bam O=22018FL-04-01-12_S12_L002_sorted_RG.bam RGID=sample RGLB=lib1 RGPL=ILLUMINA RGPU=unit1 RGSM=sample
       picard AddOrReplaceReadGroups I=22018FL-04-01-21_S21_L007_sorted.bam O=22018FL-04-01-21_S21_L007_sorted_RG.bam RGID=sample RGLB=lib1 RGPL=ILLUMINA RGPU=unit1 RGSM=sample
#      picard AddOrReplaceReadGroups I=22154FL-03-01-01_S1_L003_sorted.bam O=22154FL-03-01-01_S1_L003_sorted_RG.bam RGID=sample RGLB=lib1 RGPL=ILLUMINA RGPU=unit1 RGSM=sample

#   With the readgroups added, I added the code below to three separate slurm scripts to run. 
        picard MarkDuplicates -I 22018FL-04-01-12_S12_L002_sorted_RG.bam -O 22018FL-04-01-12_S12_L002_sorted_RG_markups.bam -M 22018FL-04-01-12_S12_L002_sorted_RG_markups_metrics.txt
        picard MarkDuplicates -I 22018FL-04-01-21_S21_L007_sorted_RG.bam -O 22018FL-04-01-21_S21_L007_sorted_RG_markups.bam -M 22018FL-04-01-21_S21_L007_sorted_RG_markups_metrics.txt
        picard MarkDuplicates -I 22154FL-03-01-01_S1_L003_sorted_RG.bam -O 22154FL-03-01-01_S1_L003_sorted_RG_markups.bam -M 22154FL-03-01-01_S1_L003_sorted_RG_markups_metrics.txt
#   Now, i have to re-run qualimap. 
        qualimap bamqc -nt 8 -bam 22018FL-04-01-12_S12_L002_sorted_RG_markups.bam --java-mem-size=16G -outdir L002_sorted_qualimap_results_R2
        qualimap bamqc -nt 8 -bam 22018FL-04-01-21_S21_L007_sorted_RG_markups.bam --java-mem-size=16G -outdir L007_sorted_qualimap_results_R2
        qualimap bamqc -nt 8 -bam 22154FL-03-01-01_S1_L003_sorted_RG_markups.bam --java-mem-size=16G -outdir L003_sorted_qualimap_results_R2
#   Saving the new directories to my local machine
            scp -r km1255@10.119.6.131:/data/Reynolds/km1255/HW4/L002_sorted_qualimap_results_R2/ .
            scp -r km1255@10.119.6.131:/data/Reynolds/km1255/HW4/L007_sorted_qualimap_results_R2/ .
            scp -r km1255@10.119.6.131:/data/Reynolds/km1255/HW4/L003_sorted_qualimap_results_R2nan/ .
